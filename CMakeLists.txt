cmake_minimum_required(VERSION 3.24)
project(TaskManagerNative VERSION 1.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Task Manager requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TASKMGR_FAVOR_SIZE "Prefer smaller release code while retaining intrinsic optimization" ON)
option(TASKMGR_LTO "Enable release link-time optimization" ON)
option(TASKMGR_DIAGNOSTICS "Include developer tests, benchmarks and screenshot tooling" OFF)
set(TASKMGR_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
if(TASKMGR_DIAGNOSTICS)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${TASKMGR_MANIFEST}")
  file(READ "${TASKMGR_MANIFEST}" TASKMGR_DEVELOPER_MANIFEST)
  string(REPLACE "requireAdministrator" "asInvoker" TASKMGR_DEVELOPER_MANIFEST "${TASKMGR_DEVELOPER_MANIFEST}")
  set(TASKMGR_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/TaskManager.developer.manifest")
  file(CONFIGURE OUTPUT "${TASKMGR_MANIFEST}" CONTENT "${TASKMGR_DEVELOPER_MANIFEST}" @ONLY)
endif()
add_executable(TaskManager WIN32 native/main.cpp native/theme.cpp native/search.cpp native/table.cpp native/performance.cpp native/commands.cpp native/monitor.cpp native/startup.cpp native/hardware.cpp native/trace.cpp native/accounts.cpp native/icons.cpp native/graph.cpp native/actions.cpp native/integration.cpp native/app.rc)
target_compile_definitions(TaskManager PRIVATE "TASKMGR_MANIFEST=\"${TASKMGR_MANIFEST}\"")
if(TASKMGR_DIAGNOSTICS)
  target_sources(TaskManager PRIVATE native/verify.cpp native/diagnostics.cpp)
  target_compile_definitions(TaskManager PRIVATE TASKMGR_DIAGNOSTICS)
endif()
target_compile_definitions(TaskManager PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(TaskManager PRIVATE comctl32 comdlg32 uxtheme dwmapi shell32 shlwapi psapi pdh iphlpapi wtsapi32 version dbghelp advapi32 ole32 oleaut32 runtimeobject xmllite taskschd gdiplus dxgi setupapi netapi32 ws2_32)
if(MSVC)
  target_compile_options(TaskManager PRIVATE /W4 /WX /permissive- /EHsc /GR- /utf-8 /Zc:__cplusplus $<$<CONFIG:Release>:/Gw> $<$<CONFIG:Release>:/Gy>)
  if(TASKMGR_LTO)
    target_compile_options(TaskManager PRIVATE $<$<CONFIG:Release>:/GL>)
    target_link_options(TaskManager PRIVATE $<$<CONFIG:Release>:/LTCG>)
  endif()
  if(TASKMGR_FAVOR_SIZE)
    target_compile_options(TaskManager PRIVATE $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/Os> $<$<CONFIG:Release>:/Oi>)
    set_source_files_properties(native/table.cpp native/graph.cpp PROPERTIES COMPILE_OPTIONS "$<$<CONFIG:Release>:/Ot>")
  else()
    target_compile_options(TaskManager PRIVATE $<$<CONFIG:Release>:/O2>)
  endif()
  target_link_options(TaskManager PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> /DYNAMICBASE /NXCOMPAT /CETCOMPAT /MANIFEST:NO)
endif()
enable_testing()
if(TASKMGR_DIAGNOSTICS)
  add_test(NAME native-self-test COMMAND TaskManager --self-test --output ${CMAKE_BINARY_DIR}/self-test.json)
  add_test(NAME native-sampler-test COMMAND TaskManager --benchmark --samples 50 --interval 0 --output ${CMAKE_BINARY_DIR}/sampler-test.json)
  add_test(NAME native-startup-inventory-test COMMAND TaskManager --startup-inventory --output ${CMAKE_BINARY_DIR}/startup-inventory-test.json)
  add_test(NAME native-ui-test COMMAND TaskManager --ui-benchmark --seconds 5 --output ${CMAKE_BINARY_DIR}/ui-test.json)
  add_test(NAME native-process-sort-stress COMMAND TaskManager --ui-benchmark --idle --sort-spam --tab 0 --seconds 5 --output ${CMAKE_BINARY_DIR}/process-sort-test.json)
  add_test(NAME native-details-sort-stress COMMAND TaskManager --ui-benchmark --idle --sort-spam --tab 5 --seconds 5 --output ${CMAKE_BINARY_DIR}/details-sort-test.json)
  add_test(NAME native-startup-sort-stress COMMAND TaskManager --ui-benchmark --idle --sort-spam --tab 3 --seconds 5 --output ${CMAKE_BINARY_DIR}/startup-sort-test.json)
  add_test(NAME native-dark-sort-stress COMMAND TaskManager --ui-benchmark --dark --idle --sort-spam --tab 5 --seconds 5 --output ${CMAKE_BINARY_DIR}/dark-sort-test.json)
  add_test(NAME native-tab-stress COMMAND TaskManager --ui-benchmark --idle --tab-spam --seconds 5 --output ${CMAKE_BINARY_DIR}/tab-test.json)
  add_test(NAME native-dark-tab-stress COMMAND TaskManager --ui-benchmark --dark --idle --tab-spam --seconds 5 --output ${CMAKE_BINARY_DIR}/dark-tab-test.json)
  set_tests_properties(native-self-test native-sampler-test native-startup-inventory-test native-ui-test native-process-sort-stress native-details-sort-stress native-startup-sort-stress native-dark-sort-stress native-tab-stress native-dark-tab-stress PROPERTIES TIMEOUT 90)
endif()

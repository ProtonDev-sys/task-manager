cmake_minimum_required(VERSION 3.24)
project(TaskManagerNative VERSION 1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Task Manager requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(TaskManager WIN32 native/main.cpp native/monitor.cpp native/icons.cpp native/graph.cpp native/actions.cpp native/diagnostics.cpp native/app.rc)
target_compile_definitions(TaskManager PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(TaskManager PRIVATE comctl32 comdlg32 uxtheme dwmapi shell32 shlwapi psapi pdh iphlpapi wtsapi32 version dbghelp advapi32 ole32 gdiplus dxgi)
if(MSVC)
  target_compile_options(TaskManager PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /Zc:__cplusplus $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
  target_link_options(TaskManager PRIVATE $<$<CONFIG:Release>:/LTCG> /DYNAMICBASE /NXCOMPAT /CETCOMPAT /MANIFEST:NO)
endif()
enable_testing()
add_test(NAME native-self-test COMMAND TaskManager --self-test --output ${CMAKE_BINARY_DIR}/self-test.json)
add_test(NAME native-ui-test COMMAND TaskManager --ui-benchmark --seconds 5 --output ${CMAKE_BINARY_DIR}/ui-test.json)
set_tests_properties(native-self-test native-ui-test PROPERTIES TIMEOUT 60)
